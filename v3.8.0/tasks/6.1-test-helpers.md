### GREEN（最小实现）
1. 创建 `tests/helpers/` 目录
2. 创建 `tests/helpers/test-utils.mjs`，ESM 导出：
   ```javascript
   import { execSync } from 'node:child_process';
   import { writeFileSync, readFileSync, existsSync, mkdirSync } from 'node:fs';
   import { join } from 'node:path';
   import { tmpdir } from 'node:os';
   import { randomUUID } from 'node:crypto';

   export function runShell(body) {
     const script = join(tmpdir(), `test-${randomUUID()}.sh`);
     writeFileSync(script, body, 'utf-8');
     return execSync(`bash ${script}`, { encoding: 'utf-8' });
   }

   export function writeMetricsLine(file, event) {
     const line = JSON.stringify({ ...event, ts: Date.now() }) + '\n';
     writeFileSync(file, line, { flag: 'a' });
   }

   export function parseFrontMatter(file) {
     const content = readFileSync(file, 'utf-8');
     const match = content.match(/^---\n([\s\S]*?)\n---/);
     if (!match) return {};
     // 简易 YAML 解析（无需引入依赖）
     const obj = {};
     for (const line of match[1].split('\n')) {
       const [key, ...rest] = line.split(':');
       if (key && rest.length) obj[key.trim()] = rest.join(':').trim();
     }
     return obj;
   }

   export function createTmpDir() {
     const dir = join(tmpdir(), `flow-kit-test-${randomUUID()}`);
     mkdirSync(dir, { recursive: true });
     return dir;
   }
   ```
