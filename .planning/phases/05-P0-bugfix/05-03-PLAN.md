```javascript
function detectPrisma(schemaPath) {
  const content = fs.readFileSync(schemaPath, 'utf8');
  const provider = content.match(/provider\s*=\s*"(\w+)"/)?.[1];
  const PROVIDER_MAP = {
    'mysql': 'mysql',
    'postgresql': 'postgresql',
    'mongodb': 'mongodb',
    'sqlite': 'sqlite',
    'sqlserver': 'mssql'
  };
  return PROVIDER_MAP[provider] || null;
}

function detectSequelize(configPath) {
  // Detect from config content
  const content = fs.readFileSync(configPath, 'utf8');
  // Look for dialect field
  const dialect = content.match(/dialect:\s*['"](\w+)['"]/)?.[1];
  return dialect || null;
}

function detectTypeORM(configPath) {
  // Detect from typeorm config
  const content = fs.readFileSync(configPath, 'utf8');
  const type = content.match(/type:\s*['"](\w+)['"]/)?.[1];
  return type || null;
}

function detectKnex(configPath) {
  // Detect from knexfile
  const content = fs.readFileSync(configPath, 'utf8');
  // Look for client: 'pg', 'mysql', etc.
  const client = content.match(/client:\s*['"](\w+)['"]/)?.[1];
  // Map common clients
  const CLIENT_MAP = { 'pg': 'postgresql', 'mysql': 'mysql', 'sqlite3': 'sqlite' };
  return CLIENT_MAP[client] || client || null;
}
```

```javascript
function writeDatabaseTypeFile(cwd, result) {
  const content = [
    `database_type: ${result.type}`,
    `detected_at: ${new Date().toISOString()}`,
    `detection_method: ${result.method || 'none'}`,
    `confidence: ${result.confidence || 'none'}`
  ].join('\n');

  const dir = path.join(cwd, '.flow-kit');
  fs.mkdirSync(dir, { recursive: true });
  fs.writeFileSync(path.join(dir, 'database-type'), content);
}
```
